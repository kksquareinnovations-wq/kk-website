## 🚀 IMPLEMENTATION EXAMPLES

### Example 1: Scroll Progress Bar
```tsx
// src/components/ScrollProgressBar.tsx
"use client";

import { useEffect, useState } from "react";

export const ScrollProgressBar = () => {
  const [progress, setProgress] = useState(0);

  useEffect(() => {
    const handleScroll = () => {
      const scrollHeight = document.documentElement.scrollHeight - window.innerHeight;
      const scrolled = (window.scrollY / scrollHeight) * 100;
      setProgress(scrolled);
    };

    window.addEventListener("scroll", handleScroll, { passive: true });
    return () => window.removeEventListener("scroll", handleScroll);
  }, []);

  return (
    <div className="fixed top-16 left-0 h-1 bg-gradient-to-r from-blue-500 to-cyan-500 z-40"
         style={{ width: `${progress}%`, transition: "width 0.1s ease-out" }} />
  );
};
```

### Example 2: Text Reveal on Scroll
```tsx
// src/components/RevealText.tsx
"use client";

import { useScrollAnimation } from "@/hooks/useScrollAnimation";

export const RevealText = ({ children, className = "" }) => {
  const { ref, isVisible } = useScrollAnimation({ threshold: 0.5 });

  return (
    <div
      ref={ref}
      className={`${isVisible ? "animate-scroll-fade-in" : "opacity-0"} ${className}`}
    >
      {children}
    </div>
  );
};
```
